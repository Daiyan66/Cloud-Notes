Create a table called Employee that contains attributes EMPNO, ENAME, JOB,
MGR, SAL and execute the following.
1. Add a column commission with domain to the Employee table.
2. Insert any five records into the table.
3. Update the column details of job.
4. Rename the column of Employ table using alter command.
5. Delete the employee whose EMPNO is 105.
-------------------------------------------------------------------------------------------------------------------
-- Step 1: Create the Employee Table
CREATE TABLE Employee (
    EMPNO INT PRIMARY KEY,
    ENAME VARCHAR(50),
    JOB VARCHAR(50),
    MGR INT,
    SAL DECIMAL(10, 2)
);

-- Step 2: Add a column COMMISSION to the Employee Table
ALTER TABLE Employee
ADD COMMISSION DECIMAL(10, 2);

-- Step 3: Insert Five Records into the Table
INSERT INTO Employee (EMPNO, ENAME, JOB, MGR, SAL, COMMISSION) VALUES
(101, 'Alice', 'Manager', NULL, 80000.00, 5000.00),
(102, 'Bob', 'Clerk', 101, 30000.00, NULL),
(103, 'Charlie', 'Analyst', 101, 50000.00, 2000.00),
(104, 'David', 'Salesman', 101, 45000.00, 2500.00),
(105, 'Eve', 'Clerk', 102, 32000.00, NULL);

-- Step 4: Update the JOB column
UPDATE Employee
SET JOB = 'Senior Manager'
WHERE EMPNO = 101;

UPDATE Employee
SET JOB = 'Senior Clerk'
WHERE JOB = 'Clerk';

-- Step 5: Rename the column ENAME to EMP_NAME
ALTER TABLE Employee
CHANGE ENAME EMP_NAME VARCHAR(50);

-- Step 6: Delete the employee whose EMPNO is 105
DELETE FROM Employee
WHERE EMPNO = 105;
