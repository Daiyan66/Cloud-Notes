#include <stdio.h>
#include <stdlib.h>
#include <time.h>

#define MAX 100000  // Define a maximum size to avoid dynamic memory allocation

void selectionSort(int arr[], int n) {
    for (int i = 0; i < n-1; i++) {
        minIdx = i;
        for (int j = i+1; j < n; j++) {
            if (arr[j] < arr[minIdx])
                minIdx = j;
        }
	int temp = arr[minIdx];
        arr[minIdx] = arr[i];
        arr[i] = temp;
    }
}

int main() {
    int n;
    int arr[MAX];
    clock_t start, end;
    double cpu_time_used;

    printf("Enter number of elements = ");
    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        arr[i] = rand()%1000;
    }
    for (int i = 0; i < n; i++) {
        printf("%d ",arr[i]);
    }

    start = clock();
    selectionSort(arr, n);
    end = clock();
    for (int i = 0; i < n; i++) {
        printf("%d ",arr[i]);
    }
    cpu_time_used = ((double)(end - start)) / CLOCKS_PER_SEC;

    printf("Time taken to sort %d elements using Selection Sort: %.4f seconds\n", n, cpu_time_used);

    return 0;
}

4. Design and implement C/C++ Program to sort a given set of n integer elements using Quick Sort method and compute its time complexity. Run the program for varied values of n> 5000 and record the time taken to sort. Plot a graph of the time taken versus n. The elements can be read from a file or can be generated using the random number generator.

#include <stdio.h>
#include <stdlib.h>
#include <time.h>


int partition(int arr[], int low, int high) {
    int pivot = arr[high];  				
    int i = low - 1;
    for (int j = low; j < high; j++) {
        if (arr[j] < pivot) {
            i++;
            int t = arr[i];
            arr[i] = arr[j];
            arr[j] = t;
        }
    }
    int t = arr[i+1];
    arr[i+1] = arr[high];
    arr[high] = t;
    return i + 1;
}

void quickSort(int arr[], int low, int high) {
    if (low < high) {
        int j = partition(arr, low, high);
        quickSort(arr, low, j - 1);
        quickSort(arr, j + 1, high);
    }
}

int main() {
    int n = 5000;  
    
    int *arr = (int *)malloc(n * sizeof(int));

    for (int i = 0; i < n; i++) {
        arr[i] = rand()%1000;
    }

    clock_t start = clock();
    quickSort(arr, 0, n - 1);
    clock_t end = clock();
     for (int i = 0; i < n; i++) {
        printf("%d ",arr[i]);
    }
    double time_taken = ((double)(end - start)) / CLOCKS_PER_SEC;

    printf("Time taken to sort %d elements: %f seconds\n", n, time_taken);

    free(arr);
    return 0;
}

5. Design and implement C/C++ Program to solve 0/1 Knapsack problem using Dynamic programming method

#include <stdio.h>

int max(int a, int b) {
    return (a > b) ? a : b;
}

int knapsack(int c, int n, int wt[], int val[]) {
    int dp[n + 1][c + 1];

    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= c; j++) {
            if (i == 0 || j == 0)
                dp[i][j] = 0;
            else if (wt[i - 1] <= j)
                dp[i][j] = max(val[i - 1] + dp[i - 1][j - wt[i - 1]], dp[i - 1][j]);
            else
                dp[i][j] = dp[i - 1][j];
        }
    }

    return dp[n][c];
}


int main() {
    int n, W, i;

    printf("Enter number of items: ");
    scanf("%d", &n);

    int val[n], wt[n];

    printf("Enter the values and weights of each item:\n");
    for (i = 0; i < n; i++) {
        printf("Item %d - Value: ", i + 1);
        scanf("%d", &val[i]);
        printf("Item %d - Weight: ", i + 1);
        scanf("%d", &wt[i]);
    }

    printf("Enter the capacity of the knapsack: ");
    scanf("%d", &W);

    int max_value = knapsack(W, wt, val, n);
    printf("\nMaximum value in the Knapsack = %d\n", max_value);

    return 0;
}
