6 a. Design and implement C/C++ Program to solve All-Pairs Shortest Paths problem using Floyd'salgorithm.

#include <stdio.h>

#define INF 99999

void floydWarshall(int V, int dist[V][V]) {

    for (int k = 0; k < V; k++) 
        for (int i = 0; i < V; i++) 
            for (int j = 0; j < V; j++) 
                if (dist[i][k] + dist[k][j] < dist[i][j])
                    dist[i][j] = dist[i][k] + dist[k][j];

    printf("Shortest distances between every pair of vertices:\n");
    for (int i = 0; i < V; i++) {
        for (int j = 0; j < V; j++) {
            if (dist[i][j] == INF)
                printf("%7s", "INF");
            else
                printf("%7d", dist[i][j]);
        }
        printf("\n");
    }
}

int main() {
    int n;
    printf("Enter the number of vertices: ");
    scanf("%d", &n);

    int graph[n][n];
    printf("Enter the adjacency matrix (use %d for INF):\n", INF);
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            scanf("%d", &graph[i][j]);

    floydWarshall(n, graph);
    return 0;
}

6 b. Design and implement C/C++ Program to find the transitive closure using Warshal's algorithm.

#include <stdio.h>

void warshall(int V, int reach[V][V]) {
  
    for (int k = 0; k < V; k++) 
        for (int i = 0; i < V; i++) 
            for (int j = 0; j < V; j++) 
                reach[i][j] = reach[i][j] || (reach[i][k] && reach[k][j]);

    printf("Transitive closure of the given graph:\n");
    for (int i = 0; i < V; i++) {
        for (int j = 0; j < V; j++)
            printf("%5d", reach[i][j]);
        printf("\n");
    }
}

int main() {
    int n;
    printf("Enter the number of vertices: ");
    scanf("%d", &n);

    int graph[n][n];
    printf("Enter the adjacency matrix (0 or 1):\n");
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            scanf("%d", &graph[i][j]);

    warshall(n, graph);
    return 0;
}

7. Design and implement C/C++ Program to find Minimum Cost Spanning Tree of a given connected undirected graph using Prim's algorithm.

#include <stdio.h>
#include <limits.h>

#define MAX 100
#define INF 9999

int main() {
    int n,min, a, b,totalCost=0,ne=0; 
    int cost[MAX][MAX];
    int visited[MAX] = {0};

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter the adjacency matrix (enter 9999 for no edge):\n");
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            scanf("%d", &cost[i][j]);

    visited[0] = 1; 

    printf("\nEdges in the Minimum Spanning Tree:\n");

    while (ne < n - 1) {
        min = INF;
        for (int i = 0; i < n; i++) 
            if (visited[i]) 
                for (int j = 0; j < n; j++) 
                    if (!visited[j] && cost[i][j] < min) {
                        min = cost[i][j];
                        a = i;
                        b = j;
                    }

        if (!visited[b]) {
            printf("Edge %d: (%d - %d) cost: %d\n", ne + 1, a, b, min);
            totalCost += min;
            visited[b] = 1;
            ne++;
        }
    }

    printf("Total cost of Minimum Spanning Tree = %d\n", totalCost);
    return 0;
}
